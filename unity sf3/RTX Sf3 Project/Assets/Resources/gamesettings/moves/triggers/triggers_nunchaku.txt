##################################################### NUNCHAKU IDLE VARIABLE
  - Name: NUNCHAKU_IDLE_VARIABLE
    Templates: [LOCK_SUBTYPE_NUNCHAKU]
    Triggers:
      - Events:
          - AnimationEnd: {Player: Me}
        Priority: 500
        Conditions:
          - Equal: {Value1: "?GetVariable[Me, Nunchaku_Idle_Weapon].Value", Value2: "1"}
          - Not:
            - Equal: {Value1: "?Player[Me].CurrentAnimation", Value2: "WEAPON"}
            - Equal: {Value1: "?Player[Me].CurrentAnimation", Value2: "STEP"}
            - Equal: {Value1: "?Player[Me].CurrentAnimation", Value2: "IDLE"}
          - Or:
            - Equal: {Value1: "?Player[Me].CurrentAnimation", Value2: "KICK"}
            - Equal: {Value1: "?Player[Me].CurrentAnimation", Value2: "MOVE"}
            - Equal: {Value1: "?Player[Me].CurrentAnimation", Value2: "BLOCK"}
            - Equal: {Value1: "?Player[Me].CurrentAnimation", Value2: "REACTION"}
            - Equal: {Value1: "?Player[Me].CurrentAnimation", Value2: "LOW_STANCE"}
        Actions:
          - SetVariable: {Name: Nunchaku_Idle_Weapon, Value: 0, ApplyTo: Me}

      - Events:
          - Birth: {Player: Me}
        Actions:
          - SetVariable: {Name: Nunchaku_Idle_Weapon, Value: 1, ApplyTo: Me}

      - Events:
          - AnimationEnd: {Player: Me}
        Priority: 500
        Conditions:
          - Not:
            - Equal: {Value1: "?GetVariable[Me, Nunchaku_Idle_Weapon].Value", Value2: "1"}
          - Equal: {Value1: "?Player[Me].CurrentAnimation", Value2: "WEAPON"}
        Actions:
          - SetVariable: {Name: Nunchaku_Idle_Weapon, Value: 1, ApplyTo: Me}

      - Events:
          - Birth: {Player: Me}
        Priority: -99999
        Actions:
          - SetVariable: {Name: Nunchaku_Idle_Weapon, Value: 1, ApplyTo: Me}
